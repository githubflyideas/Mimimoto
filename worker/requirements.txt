# The worker itself needs nothing beyond the standard library; everything here
# is CosyVoice 2's own dependency surface.
#
# Install CosyVoice from its repository rather than pinning it here — it is not
# published to PyPI, and it needs its Matcha-TTS submodule on PYTHONPATH:
#
#   git clone --recursive https://github.com/FunAudioLLM/CosyVoice.git
#   pip install -r CosyVoice/requirements.txt
#   export PYTHONPATH="$PWD/CosyVoice:$PWD/CosyVoice/third_party/Matcha-TTS"
#
#   pip install modelscope
#   modelscope download --model iic/CosyVoice2-0.5B \
#       --local_dir pretrained_models/CosyVoice2-0.5B
#
# Apache-2.0, commercial use permitted, no carve-out on the cloning feature —
# which is why it replaced FireRedTTS3. See docs/DECISIONS.md D-015.
#
